from processing import *

def setup():
  size(500,500)

def buildings():
  fill(100)
  rect(0,250,100,250)
  fill(50)
  rect(15,275,65,75)
  fill(100)
  rect(100,200,150,300)
  fill(50)
  rect(125,225,50,50)
  rect(175,225,50,50)
  rect(125,275,50,50)
  rect(175,275,50,50)
  fill(100)
  rect(250,300,125,200)
  fill(100)
  rect(375,50,125,450)
  fill(50)
  rect(400,75,100,50)
  rect(400,175,100,50)
  fill(249, 224, 158)
  rect(400,275,100,50)
  
def draw_rain_drops(n):
  i = 0;
  while i < n: 
    x = random(0,500)
    y1 = random(0,500)
    y2 = random(y1+25,y1+50)
    line(x,y1,x,y2)
    i = i + 1 

def draw():
  background(125)
  buildings()
  draw_rain_drops(100)

run()
