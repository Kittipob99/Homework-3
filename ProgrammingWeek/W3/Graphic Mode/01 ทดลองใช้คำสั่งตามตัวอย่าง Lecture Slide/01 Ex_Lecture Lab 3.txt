from math import sin
from processing import *

#โค้ดนี้ถูกเขียนบนเว็บไซต์  trinket.io

X = 100 #ตำแหน่งแกน X ของลูกโป่ง
Y = 600 #ตำแหน่งแกน Y ของลูกโป่ง
fly = 0 #ตัวแปรที่นำไปลบกับค่า Y เพื่อทำให้ลูกโป่งลอยขึ้น

def setup():
  size(500,500)

def balloon(X,Y): #กำหนดฟังก์ชันเพื่อใช้วาดลูกโป่ง
  ellipse(X,Y-fly,50,75)
  line(X,(Y+37.5)-fly,X,(Y+112.5)-fly)

def draw_balloons():
  global X,Y,fly
  background(100+sin(fly*0.01)*100) #นำตรีโกณมาใช้เพื่อให้สีพื้นหลังปรับเปลี่ยนตลอดเวลาตามค่า fly
  fly = fly + 2.5
  fill(random(0,255),0,0) #สุ่มความเข้มของลูกโป่งสีแดงตลอดเวลา
  balloon(X,Y)
  fill(0,random(0,255),0) #สุ่มความเข้มของลูกโป่งสีเขียวตลอดเวลา
  balloon(X+150,Y)
  fill(0,0,random(0,255)) #สุ่มความเข้มของลูกโป่งสีฟ้าตลอดเวลา
  balloon(X+300,Y)
  if Y - fly == -100: #ถ้าลูกโป่งออกจากหน้าจอแล้ว ให้กลับมาที่เดิม
    Y = 600
    fly = 0

draw = draw_balloons
run()
