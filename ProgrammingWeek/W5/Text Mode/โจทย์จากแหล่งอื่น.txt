#หาค่าซ้ำใน Array แล้วลบทิ้ง
a = [3, 1, 4, 2, 3, 1]

def find_copy(array):
    i = 0
    n = i + 1
    while i < len(array):
        while n < len(array):
            if array[i] == array[n]:
                del array[n]
            n += 1
        i += 1
        n = i + 1
    return array

print(find_copy(a))
