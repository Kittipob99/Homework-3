from processing import *

def setup():
  size(500,500)

x = [37.5,137.5,237.5,337.5,437.5]
y = [250,250,250,250,250]
big = [25,25,25,25,25]
  
def draw_positive(x,y,size):
  rect(x,y,size,size)
  rect(x+size,y,size,size)
  rect(x-size,y,size,size)
  rect(x,y+size,size,size)
  rect(x,y-size,size,size)

def draw():
  global x,y,big
  i = 0
  n = 0
  background(125)
  while i < len(x):
    draw_positive(x[n],y[n],big[n])
    if -50 < x[n] - mouseX < 25 and -50 < y[n] - mouseY < 25:
      big[n] = 35
    else:
      big[n] = 25
    n += 1
    i += 1
  
run()
