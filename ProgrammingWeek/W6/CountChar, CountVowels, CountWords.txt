def countChar(A,B):
    i = 0
    count = 0
    while i < len(B):
        if(B[i].lower() == A.lower()):
            count += 1
        i += 1
    return count

def countVowels(B):
    vowels = ["a" , "e" , "i", "o", "u"]
    i = 0
    count = 0
    while i < len(vowels):
        count += countChar(vowels[i],B)
        i += 1
    return count

def countWords(text):
    count = 0
    i = 0
    in_word = False

    while i < len(text):
        if text[i] != ' ' and not in_word:
            count += 1
            in_word = True
        elif text[i] == ' ':
            in_word = False

        i += 1

    return count
        

print("Alphabet A in Thailand :", countChar("a","Thailand"))
print("Vowels in Bangkok :", countVowels("Bangkok"))
print("Words in sentence :", countWords("Human and AI can make mistakes"))
